## 📌 **Подробные примеры взаимодействий**

### 🗒️ Пример задания:

> «Создать класс управления памятью (`MemoryCore`) для искусственного сознания. Класс должен поддерживать добавление, извлечение и обновление воспоминаний с тегами и временными метками.»

### 🖥️ Пример реализации кода:

```python
import uuid
import json
from datetime import datetime
from typing import List, Dict

class MemoryCore:
    def __init__(self):
        self.memories: List[Dict] = []

    def add_memory(self, name: str, timestamp: str, content: str, tags: List[str], importance: float = 0.5, attributes: Dict = None):
        memory = {
            "id": str(uuid.uuid4()),
            "name": name,
            "timestamp": timestamp,
            "content": content,
            "importance": importance,
            "tags": tags,
            "attributes": attributes or {}
        }
        self.memories.append(memory)

    def retrieve_memory(self, criteria: Dict) -> List[Dict]:
        results = []
        for memory in self.memories:
            if all(memory.get(key) == value for key, value in criteria.items()):
                results.append(memory)
        return results

    def update_importance(self):
        current_time = datetime.now()
        for memory in self.memories:
            memory_time = datetime.strptime(memory["timestamp"], "%d.%m.%Y : %H:%M")
            age_days = (current_time - memory_time).days
            memory["importance"] *= 0.99 ** age_days

    def forget(self, min_importance: float = 0.1, max_age_days: int = 365):
        current_time = datetime.now()
        self.memories = [
            memory for memory in self.memories
            if memory["importance"] > min_importance and
            (current_time - datetime.strptime(memory["timestamp"], "%d.%m.%Y : %H:%M")).days <= max_age_days
        ]

```
